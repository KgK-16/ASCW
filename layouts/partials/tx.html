{{- /* One text field of a block, in the page's language. b = block, f = field prefix ("" → en / ru, "cap_" → cap_en / cap_ru), tag = element, cls = optional class, d = default {en, ru} when the block has no text of its own. A missing language falls back to the other one (marked with its lang). The stored HTML is already cleaned by the editor. */ -}} {{- $en := index .b (print .f "en") }}{{ $ru := index .b (print .f "ru") -}} {{- if and (not (isset .b (print .f "en"))) (not (isset .b (print .f "ru"))) .d }}{{ $en = .d.en }}{{ $ru = .d.ru }}{{ end -}} {{- $inRu := eq site.Language.Name "ru" -}} {{- $text := cond $inRu ($ru | default $en) ($en | default $ru) -}} {{- if $text -}} {{- $class := cond (and (isset . "cls") .cls) (printf ` class="%s"` .cls) "" -}} {{- $lang := cond (cond $inRu $ru $en) "" (printf ` lang="%s"` (cond $inRu "en" "ru")) -}} {{- printf `<%s%s%s>%s` .tag $lang $class $text .tag | safeHTML -}} {{- end -}}

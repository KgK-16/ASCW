" " -}} {{- if in (slice "left" "right" "center" "full") $m }}{{ $align = $m -}} {{- else if findRE `^[0-9]+%$` $m }}{{ $width = $m -}} {{- else if findRE `^[0-9]+$` $m }}{{ $width = print $m "px" -}} {{- else if $m }}{{ $caption = $m }}{{ end -}} {{- end -}}
